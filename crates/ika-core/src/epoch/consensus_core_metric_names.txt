# Snapshot of every prometheus metric name registered by upstream Sui's
# `consensus-core` crate (mainnet-v1.76.1), from consensus/core/src/metrics.rs
# and consensus/core/src/network/metrics.rs.
#
# WHY: ika starts consensus-core with a namespaced registry —
# `Registry::new_custom(Some("ika_consensus"))` in consensus_manager/mod.rs —
# so every name below is exported as `ika_consensus_<name>`. ika ALSO registers
# its own `ika_consensus_*` names into a different registry, and RegistryService
# merges both at the /metrics endpoint. A name present on both sides is served
# twice and Prometheus drops one sample per scrape (ika #2022).
#
# USED BY: the `ika_consensus_namespace_does_not_collide_with_consensus_core`
# test in crates/ika-core/src/epoch/epoch_metrics.rs.
#
# REGENERATE after a Sui bump (from the consensus/core/src dir of the pinned
# sui checkout):
#   python3 -c 'import pathlib,re;M=re.compile(r"register_[a-z_]+_with_registry!\s*\(\s*\"([a-z0-9_]+)\"",re.S);print("\n".join(sorted({m.group(1) for p in pathlib.Path(".").rglob("*.rs") for m in M.finditer(p.read_text(errors="replace"))})))'
